# ========================================
#   CONFIGURACAO DO SERVIX
# ========================================

# 1. BANCO DE DADOS (PostgreSQL)
# Crie um arquivo .env na pasta api_servix com:
DB_HOST=localhost
DB_PORT=5432
DB_NAME=servix
DB_USER=seu_usuario_postgres
DB_PASS=sua_senha_postgres
JWT_SECRET=seu_jwt_secret_aqui_123456
PORT=3000

# 2. FIREBASE (Frontend Web)
# Crie um arquivo .env.local na pasta servix-web com:
REACT_APP_FIREBASE_API_KEY=sua_api_key_aqui
REACT_APP_FIREBASE_AUTH_DOMAIN=seu-projeto.firebaseapp.com
REACT_APP_FIREBASE_PROJECT_ID=seu-projeto-id
REACT_APP_FIREBASE_STORAGE_BUCKET=seu-projeto.appspot.com
REACT_APP_FIREBASE_MESSAGING_SENDER_ID=123456789
REACT_APP_FIREBASE_APP_ID=1:123456789:web:abcdef123456

# 3. FIREBASE (Mobile)
# Copie as mesmas configurações para src/firebase.js

# ========================================
#   COMANDOS PARA EXECUTAR
# ========================================

# Backend:
# cd api_servix
# npm start

# Frontend Web:
# cd servix-web
# npm start

# Mobile:
# cd servix-app
# npx expo start

# ========================================
#   SOLUCAO DE PROBLEMAS
# ========================================

# Se der erro "Missing script: start":
# cd servix-web
# npm install
# npm start

# Se der erro "react-scripts não é reconhecido":
# cd servix-web
# npm install react-scripts
# npm start

# Se der erro "Module not found":
# rm -rf node_modules package-lock.json
# npm install

# Se der erro "Invalid options object":
# cd servix-web
# rm -rf node_modules package-lock.json
# npm install
# npm start
